    72	function readMicrolinkUrl(
    73	  value: MicrolinkValue,

    77	  }

    79	  if (
    80	    value &&
    81	    typeof value === 'object' &&

    90	function normalizeMicrolinkPreview(
    91	  url: string,
    92	  label: string | undefined,
    93	  username: string | undefined,
    94	  payload: unknown,
    95	): LinkPreviewData {
    96	  if (
    97	    !payload ||
    98	    typeof payload !== 'object' ||
    99	    !('data' in payload)
   100	  ) {
   101	    throw new Error(
   102	      'Preview service returned invalid data.',
   103	    )
   104	  }
   105	
   106	  const data = (
   107	    payload as {
   108	      data?: Record<string, unknown>
   109	    }
   110	  ).data
   111	
   112	  if (!data) {
   113	    throw new Error(
   114	      'Preview service returned no metadata.',
   115	    )
   116	  }
   117	
   118	  const detected =
   119	    detectPlatform(
   120	      url,
   121	      label,
   122	    )
   123	
   124	  const platform:
   125	    LinkPreviewData['platform'] =
   126	    detected === 'photography'
   127	      ? 'web'
   128	      : detected
   129	
   130	  let hostname = url
   131	
   132	  try {
   133	    hostname =
   134	      new URL(url).hostname.replace(
   135	        /^www\./,
   136	        '',
   137	      )
   138	  } catch {
   139	    // Keep original URL.
   140	  }
   141	
   142	  const title =
   143	    typeof data.title === 'string'
   144	      ? data.title
   145	      : label ||
   146	        username ||
   147	        hostname
   148	
   149	  const description =
   150	    typeof data.description === 'string'
   151	      ? data.description
   152	      : `Preview for ${hostname}.`
   153	
   154	  const siteName =
   155	    typeof data.publisher === 'string'
   156	      ? data.publisher
   157	      : typeof data.siteName === 'string'
   158	        ? data.siteName
   159	        : username ||
   160	          hostname
   161	
   162	  const image =
   163	    readMicrolinkUrl(
   164	      data.image as MicrolinkValue,
   165	    )
   166	
   167	  const favicon =
   168	    readMicrolinkUrl(
   169	      data.logo as MicrolinkValue,
   170	    ) ||
   171	    readMicrolinkUrl(
   172	      data.favicon as MicrolinkValue,
   173	    )
   174	
   175	  return {
   176	    ok: true,
   177	    url,
   178	    platform,
   179	    title,
   180	    description,
   181	    siteName,
   182	    favicon,
   183	    images: image
   184	      ? [image]
   185	      : [],
   186	    image: image || null,
   187	  }
   188	}

   195	function detectPlatform(
   196	  url: string,
   197	  label?: string,
   198	): PlatformKey {
   199	  const value = `${url} ${label || ''}`.toLowerCase()
   200	
   201	  if (
   202	    value.includes('instagram.com') ||
   203	    value.includes('instagram')
   204	  ) {
   205	    return 'instagram'
   206	  }
   207	
   208	  if (
   209	    value.includes('github.com') ||
   210	    value.includes('github')
   211	  ) {
   212	    return 'github'
   213	  }
   214	
   215	  if (
   216	    value.includes('linkedin.com') ||
   217	    value.includes('linkedin')
   218	  ) {
   219	    return 'linkedin'
   220	  }
   221	
   222	  if (
   223	    value.includes('youtube.com') ||
   224	    value.includes('youtu.be') ||
   225	    value.includes('youtube')
   226	  ) {
   227	    return 'youtube'
   228	  }
   229	
   230	  if (
   231	    value.includes('x.com') ||
   232	    value.includes('twitter.com') ||
   233	    value.includes(' twitter')
   234	  ) {
   235	    return 'x'
   236	  }
   237	
   238	  if (
   239	    value.includes('photography') ||
   240	    value.includes('camera')
   241	  ) {
   242	    return 'photography'
   243	  }
   244	
   245	  return 'web'
   246	}

   333	export function RichLinkPreview({
   334	  url,
   335	  label,
   336	  username,
   337	  accent,
   338	}: RichLinkPreviewProps) {
   339	  const [data, setData] =
   340	    useState<LinkPreviewData | null>(null)
   341	
   342	  const [loading, setLoading] =
   343	    useState(true)
   344	
   345	  const [error, setError] =
   346	    useState<string | null>(null)
   347	
   348	  useEffect(() => {
   349	    let cancelled = false
   350	
   351	    setLoading(true)
   352	    setError(null)
   353	    setData(null)
   354	
   355	    fetch(buildApiUrl(url))
   356	      .then(async (response) => {
   357	        let payload: unknown
   358	
   359	        try {
   360	          payload = await response.json()
   361	        } catch {
   362	          throw new Error(
   363	            'Preview service returned invalid JSON.',
   364	          )
   365	        }
   366	
   367	        if (!response.ok) {
   368	          throw new Error(
   369	            'Preview request failed.',
   370	          )
   371	        }
   372	
   373	        return normalizeMicrolinkPreview(
   374	          url,
   375	          label,
   376	          username,
   377	          payload,
   378	        )
   379	      })
   380	      .then((result) => {
   381	        if (cancelled) {
   382	          return
   383	        }
   384	
   385	        setData(result)
   386	      })
   387	      .catch((reason) => {
   388	        if (cancelled) {
   389	          return
   390	        }
   391	
   392	        setError(
   393	          reason instanceof Error
   394	            ? reason.message
   395	            : 'Preview unavailable.',
   396	        )
   397	      })
   398	      .finally(() => {
   399	        if (!cancelled) {
   400	          setLoading(false)
   401	        }
   402	      })
   403	
   404	    return () => {
   405	      cancelled = true
   406	    }
   407	  }, [
   408	    url,
   409	    label,
   410	    username,
   411	  ])
   412	
   413	  const detectedPlatform = useMemo(
   414	    () =>
   415	      detectPlatform(
   416	        data?.url || url,
   417	        label,
   418	      ),
   419	    [data?.url, label, url],
   420	  )
   421	
   422	  const platform =
   423	    data?.platform || detectedPlatform
   424	
   425	  const platformInfo = getPlatformInfo(
   426	    platform as PlatformKey,
   427	  )
   428	
   429	  const images = getPreviewImages(data)
   430	
   431	  const displayTitle =
   432	    data?.title ||
   433	    label ||
   434	    platformInfo.name
   435	
   436	  const displayUsername =
   437	    username ||
   438	    data?.siteName ||
   439	    platformInfo.name
   440	
   441	  const hasPreview =
   442	    !error &&
   443	    !loading &&
   444	    Boolean(data)
   445	
   446	  const rootStyle = {
   447	    '--preview-accent':
   448	      accent || platformInfo.color,
   449	    '--platform-color':
   450	      platformInfo.color,
   451	  } as CSSProperties
   452	
   453	  return (
   454	    <div
   455	      className={`${styles.root} ${
   456	        loading
   457	          ? styles.rootLoading
   458	          : hasPreview
   459	            ? styles.rootPreview
   460	            : styles.rootFallback
   461	      }`}
   462	      style={rootStyle}
   463	    >
   464	      {/* =====================================================
   465	          PROFILE HEADER
   466	      ===================================================== */}
   467	
   468	      <div className={styles.identity}>
   469	        <div className={styles.avatar}>
   470	          {data?.image ? (
   471	            <img
   472	              src={data.image}
   473	              alt=""
   474	              loading="lazy"
   475	              onError={(event) => {
   476	                event.currentTarget.style.display =
   477	                  'none'
   478	              }}
   479	            />
   480	          ) : data?.favicon ? (
   481	            <img
   482	              src={data.favicon}
   483	              alt=""
   484	              loading="lazy"
   485	            />
   486	          ) : (
   487	            <span>
   488	              {platformInfo.icon}
   489	            </span>
   490	          )}
   491	        </div>
   492	
   493	        <div className={styles.identityText}>
   494	          <strong>
   495	            {displayUsername}
   496	          </strong>
   497	
   498	          <span>
   499	            {displayTitle}
   500	          </span>
   501	        </div>
   502	
   503	        <span className={styles.platform}>
   504	          {data
   505	            ? getPlatformInfo(
   506	                data.platform,
   507	              ).name
   508	            : platformInfo.name}
   509	        </span>
   510	      </div>
   511	
   512	      {/* =====================================================
   513	          PREVIEW / FALLBACK CONTAINER
   514	      ===================================================== */}
   515	
   516	      <div className={styles.previewContainer}>
   517	        {loading && (
   518	          <div
   519	            className={styles.loadingState}
   520	            aria-label="Loading preview"
   521	          >
   522	            <div
   523	              className={
   524	                styles.loadingOrb
   525	              }
   526	            />
   527	
   528	            <span>
   529	              FETCHING PREVIEW
   530	            </span>
   531	          </div>
   532	        )}
   533	
   534	        {!loading &&
   535	          hasPreview &&
   536	          images.length > 0 && (
   537	            <PreviewMedia
   538	              images={images}
   539	              title={displayTitle}
   540	            />
   541	          )}
   542	
   543	        {!loading &&
   544	          hasPreview &&
   545	          images.length === 0 && (

   547	              title={displayTitle}
   548	              description={
   549	                data?.description || ''

   553	          )}
   554	
   555	        {!loading &&
   556	          !hasPreview && (
   557	            <FallbackPreview
   558	              platform={platformInfo}
   559	            />
   560	          )}
   561	      </div>
   562	
   563	      {/* =====================================================
   564	          FOOTER
   565	      ===================================================== */}
   566	
   567	      <div className={styles.footer}>
   568	        <div className={styles.footerInfo}>
   569	          <span>
   570	            {data?.siteName ||
   571	              label ||
   572	              platformInfo.name}
   573	          </span>

   575	          <small>

   579	              : 'Preview unavailable — open the original page'}

   581	        </div>

   583	        <motion.span

   591	      </div>
   592	    </div>
   593	  )
   594	}

   600	function PreviewMedia({
   601	  images,
   602	  title,
   603	}: {
   604	  images: string[]
   605	  title: string
   606	}) {
   607	  const visibleImages = images.slice(0, 5)
   608	
   609	  const [failedImages, setFailedImages] =
   610	    useState<string[]>([])
   611	
   612	  const validImages =
   613	    visibleImages.filter(
   614	      (image) =>
   615	        !failedImages.includes(image),
   616	    )
   617	
   618	  const markFailed = (
   619	    image: string,
   620	  ) => {
   621	    setFailedImages((current) =>
   622	      current.includes(image)
   623	        ? current
   624	        : [...current, image],
   625	    )
   626	  }
   627	
   628	  if (validImages.length === 0) {
   629	    return (
   630	      <div className={styles.imageFailure}>
   631	        <span>
   632	          PREVIEW IMAGE UNAVAILABLE
   633	        </span>
   634	      </div>
   635	    )
   636	  }
   637	
   638	  return (
   639	    <div className={styles.mediaStage}>
   640	      <div
   641	        className={styles.primaryImage}
   642	      >
   643	        <img
   644	          src={validImages[0]}
   645	          alt={title}
   646	          loading="lazy"
   647	          onError={() =>
   648	            markFailed(validImages[0])
   649	          }
   650	        />
   651	      </div>
   652	
   653	      {validImages
   654	        .slice(1, 5)
   655	        .map((image, index) => (
   656	          <div
   657	            key={image}
   658	            className={`${styles.floatingImage} ${
   659	              styles[
   660	                `floatingImage${index + 1}`
   661	              ]
   662	            }`}
   663	          >
   664	            <img
   665	              src={image}
   666	              alt=""
   667	              loading="lazy"
   668	              onError={() =>
   669	                markFailed(image)
   670	              }
   671	            />
   672	          </div>
   673	        ))}
   674	
   675	      <div
   676	        className={styles.mediaBadge}
   677	      >
   678	        LIVE PREVIEW
   679	      </div>
   680	    </div>
   681	  )
   682	}
