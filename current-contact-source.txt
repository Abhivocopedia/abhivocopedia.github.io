    66	function readMicrolinkUrl(
    67	  value: MicrolinkValue,
    68	) {
    69	  if (typeof value === 'string') {
    70	    return value
    71	  }
    72	
    73	  if (
    74	    value &&
    75	    typeof value === 'object' &&
    76	    typeof value.url === 'string'

    84	function normalizeMicrolinkPreview(
    85	  url: string,
    86	  label: string | undefined,
    87	  username: string | undefined,
    88	  payload: unknown,
    89	): LinkPreviewData {
    90	  if (
    91	    !payload ||
    92	    typeof payload !== 'object' ||
    93	    !('data' in payload)
    94	  ) {
    95	    throw new Error(
    96	      'Preview service returned invalid data.',
    97	    )
    98	  }
    99	
   100	  const data = (
   101	    payload as {
   102	      data?: Record<string, unknown>
   103	    }
   104	  ).data
   105	
   106	  if (!data) {
   107	    throw new Error(
   108	      'Preview service returned no metadata.',
   109	    )
   110	  }
   111	
   112	  const detected =
   113	    detectPlatform(
   114	      url,
   115	      label,
   116	    )
   117	
   118	  const platform:
   119	    LinkPreviewData['platform'] =
   120	    detected === 'photography'
   121	      ? 'web'
   122	      : detected
   123	
   124	  let hostname = url
   125	
   126	  try {
   127	    hostname =
   128	      new URL(url).hostname.replace(
   129	        /^www\./,
   130	        '',
   131	      )
   132	  } catch {
   133	    // Keep original URL.
   134	  }
   135	
   136	  const title =
   137	    typeof data.title === 'string'
   138	      ? data.title
   139	      : label ||
   140	        username ||
   141	        hostname
   142	
   143	  const description =
   144	    typeof data.description === 'string'
   145	      ? data.description
   146	      : `Preview for ${hostname}.`
   147	
   148	  const siteName =
   149	    typeof data.publisher === 'string'
   150	      ? data.publisher
   151	      : typeof data.siteName === 'string'
   152	        ? data.siteName
   153	        : username ||
   154	          hostname
   155	
   156	  const image =
   157	    readMicrolinkUrl(
   158	      data.image as MicrolinkValue,
   159	    )
   160	
   161	  const favicon =
   162	    readMicrolinkUrl(
   163	      data.logo as MicrolinkValue,
   164	    ) ||
   165	    readMicrolinkUrl(
   166	      data.favicon as MicrolinkValue,
   167	    )
   168	
   169	  return {
   170	    ok: true,
   171	    url,
   172	    platform,
   173	    title,
   174	    description,
   175	    siteName,
   176	    favicon,
   177	    images: image
   178	      ? [image]
   179	      : [],
   180	    image: image || null,
   181	  }
   182	}

   189	function detectPlatform(
   190	  url: string,
   191	  label?: string,
   192	): PlatformKey {
   193	  const value = `${url} ${label || ''}`.toLowerCase()
   194	
   195	  if (
   196	    value.includes('instagram.com') ||
   197	    value.includes('instagram')
   198	  ) {
   199	    return 'instagram'
   200	  }
   201	
   202	  if (
   203	    value.includes('github.com') ||
   204	    value.includes('github')
   205	  ) {
   206	    return 'github'
   207	  }
   208	
   209	  if (
   210	    value.includes('linkedin.com') ||
   211	    value.includes('linkedin')
   212	  ) {
   213	    return 'linkedin'
   214	  }
   215	
   216	  if (
   217	    value.includes('youtube.com') ||
   218	    value.includes('youtu.be') ||
   219	    value.includes('youtube')
   220	  ) {
   221	    return 'youtube'
   222	  }
   223	
   224	  if (
   225	    value.includes('x.com') ||
   226	    value.includes('twitter.com') ||
   227	    value.includes(' twitter')
   228	  ) {
   229	    return 'x'
   230	  }
   231	
   232	  if (
   233	    value.includes('photography') ||
   234	    value.includes('camera')
   235	  ) {
   236	    return 'photography'
   237	  }
   238	
   239	  return 'web'
   240	}

   327	export function RichLinkPreview({
   328	  url,
   329	  label,
   330	  username,
   331	  accent,
   332	}: RichLinkPreviewProps) {
   333	  const [data, setData] =
   334	    useState<LinkPreviewData | null>(null)
   335	
   336	  const [loading, setLoading] =
   337	    useState(true)
   338	
   339	  const [error, setError] =
   340	    useState<string | null>(null)
   341	
   342	  useEffect(() => {
   343	    let cancelled = false
   344	
   345	    setLoading(true)
   346	    setError(null)
   347	    setData(null)
   348	
   349	    fetch(buildApiUrl(url))
   350	      .then(async (response) => {
   351	        let payload: unknown
   352	
   353	        try {
   354	          payload = await response.json()
   355	        } catch {
   356	          throw new Error(
   357	            'Preview service returned invalid JSON.',
   358	          )
   359	        }
   360	
   361	        if (!response.ok) {
   362	          throw new Error(
   363	            'Preview request failed.',
   364	          )
   365	        }
   366	
   367	        return normalizeMicrolinkPreview(
   368	          url,
   369	          label,
   370	          username,
   371	          payload,
   372	        )
   373	      })
   374	      .then((result) => {
   375	        if (cancelled) {
   376	          return
   377	        }
   378	
   379	        setData(result)
   380	      })
   381	      .catch((reason) => {
   382	        if (cancelled) {
   383	          return
   384	        }
   385	
   386	        setError(
   387	          reason instanceof Error
   388	            ? reason.message
   389	            : 'Preview unavailable.',
   390	        )
   391	      })
   392	      .finally(() => {
   393	        if (!cancelled) {
   394	          setLoading(false)
   395	        }
   396	      })
   397	
   398	    return () => {
   399	      cancelled = true
   400	    }
   401	  }, [
   402	    url,
   403	    label,
   404	    username,
   405	  ])
   406	
   407	  const detectedPlatform = useMemo(
   408	    () =>
   409	      detectPlatform(
   410	        data?.url || url,
   411	        label,
   412	      ),
   413	    [data?.url, label, url],
   414	  )
   415	
   416	  const platform =
   417	    data?.platform || detectedPlatform
   418	
   419	  const platformInfo = getPlatformInfo(
   420	    platform as PlatformKey,
   421	  )
   422	
   423	  const images = getPreviewImages(data)
   424	
   425	  const displayTitle =
   426	    data?.title ||
   427	    label ||
   428	    platformInfo.name
   429	
   430	  const displayUsername =
   431	    username ||
   432	    data?.siteName ||
   433	    platformInfo.name
   434	
   435	  const hasPreview =
   436	    !error &&
   437	    !loading &&
   438	    Boolean(data)
   439	
   440	  const rootStyle = {
   441	    '--preview-accent':
   442	      accent || platformInfo.color,
   443	    '--platform-color':
   444	      platformInfo.color,
   445	  } as CSSProperties
   446	
   447	  return (
   448	    <div
   449	      className={`${styles.root} ${
   450	        loading
   451	          ? styles.rootLoading
   452	          : hasPreview
   453	            ? styles.rootPreview
   454	            : styles.rootFallback
   455	      }`}
   456	      style={rootStyle}
   457	    >
   458	      {/* =====================================================
   459	          PROFILE HEADER
   460	      ===================================================== */}
   461	
   462	      <div className={styles.identity}>
   463	        <div className={styles.avatar}>
   464	          {data?.image ? (
   465	            <img
   466	              src={data.image}
   467	              alt=""
   468	              loading="lazy"
   469	              onError={(event) => {
   470	                event.currentTarget.style.display =
   471	                  'none'
   472	              }}
   473	            />
   474	          ) : data?.favicon ? (
   475	            <img
   476	              src={data.favicon}
   477	              alt=""
   478	              loading="lazy"
   479	            />
   480	          ) : (
   481	            <span>
   482	              {platformInfo.icon}
   483	            </span>
   484	          )}
   485	        </div>
   486	
   487	        <div className={styles.identityText}>
   488	          <strong>
   489	            {displayUsername}
   490	          </strong>
   491	
   492	          <span>
   493	            {displayTitle}
   494	          </span>
   495	        </div>
   496	
   497	        <span className={styles.platform}>
   498	          {data
   499	            ? getPlatformInfo(
   500	                data.platform,
   501	              ).name
   502	            : platformInfo.name}
   503	        </span>
   504	      </div>
   505	
   506	      {/* =====================================================
   507	          PREVIEW / FALLBACK CONTAINER
   508	      ===================================================== */}
   509	
   510	      <div className={styles.previewContainer}>
   511	        {loading && (
   512	          <div
   513	            className={styles.loadingState}
   514	            aria-label="Loading preview"
   515	          >
   516	            <div
   517	              className={
   518	                styles.loadingOrb
   519	              }
   520	            />
   521	
   522	            <span>
   523	              FETCHING PREVIEW
   524	            </span>
   525	          </div>
   526	        )}
   527	
   528	        {!loading &&
   529	          hasPreview &&
   530	          images.length > 0 && (
   531	            <PreviewMedia
   532	              images={images}
   533	              title={displayTitle}
   534	            />
   535	          )}
   536	
   537	        {!loading &&
   538	          hasPreview &&
   539	          images.length === 0 && (
   540	            <PreviewContent
   541	              title={displayTitle}
   542	              description={
   543	                data?.description || ''
   544	              }
   545	              favicon={data?.favicon}
   546	            />
   547	          )}
   548	
   549	        {!loading &&
   550	          !hasPreview && (
   551	            <FallbackPreview
   552	              platform={platformInfo}
   553	            />
   554	          )}
   555	      </div>
   556	
   557	      {/* =====================================================
   558	          FOOTER
   559	      ===================================================== */}
   560	
   561	      <div className={styles.footer}>
   562	        <div className={styles.footerInfo}>
   563	          <span>
   564	            {data?.siteName ||
   565	              label ||
   566	              platformInfo.name}
   567	          </span>
   568	
   569	          <small>
   570	            {hasPreview
   571	              ? data?.description ||
   572	                'Public page preview'
   573	              : 'Preview unavailable — open the original page'}
   574	          </small>
   575	        </div>
   576	
   577	        <motion.span
   578	          className={styles.open}
   579	          whileHover={{
   580	            x: 4,
   581	          }}
   582	        >
   583	          OPEN ↗
   584	        </motion.span>
   585	      </div>
   586	    </div>
   587	  )
   588	}

   590	/* =========================================================
   591	   REAL IMAGE PREVIEW
   592	========================================================= */
   593	
   594	function PreviewMedia({
   595	  images,
   596	  title,
   597	}: {
   598	  images: string[]
   599	  title: string
   600	}) {
   601	  const visibleImages = images.slice(0, 5)
   602	
   603	  const [failedImages, setFailedImages] =
   604	    useState<string[]>([])
   605	
   606	  const validImages =
   607	    visibleImages.filter(
   608	      (image) =>
   609	        !failedImages.includes(image),
   610	    )
   611	
   612	  const markFailed = (
   613	    image: string,
   614	  ) => {
   615	    setFailedImages((current) =>
   616	      current.includes(image)
   617	        ? current
   618	        : [...current, image],
   619	    )
   620	  }
   621	
   622	  if (validImages.length === 0) {
   623	    return (
   624	      <div className={styles.imageFailure}>
   625	        <span>
   626	          PREVIEW IMAGE UNAVAILABLE
   627	        </span>
   628	      </div>
   629	    )
   630	  }
   631	
   632	  return (
   633	    <div className={styles.mediaStage}>
   634	      <div
   635	        className={styles.primaryImage}
   636	      >
   637	        <img
   638	          src={validImages[0]}
   639	          alt={title}
   640	          loading="lazy"
   641	          onError={() =>
   642	            markFailed(validImages[0])
   643	          }
   644	        />
   645	      </div>
   646	
   647	      {validImages
   648	        .slice(1, 5)
   649	        .map((image, index) => (
   650	          <div
   651	            key={image}
   652	            className={`${styles.floatingImage} ${
   653	              styles[
   654	                `floatingImage${index + 1}`
   655	              ]
   656	            }`}
   657	          >
   658	            <img
   659	              src={image}
   660	              alt=""
   661	              loading="lazy"
   662	              onError={() =>
   663	                markFailed(image)
   664	              }
   665	            />
   666	          </div>
   667	        ))}
   668	
   669	      <div
   670	        className={styles.mediaBadge}
   671	      >
   672	        LIVE PREVIEW
   673	      </div>
   674	    </div>
   675	  )
   676	}
